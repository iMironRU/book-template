\documentclass[$if(fontsize)$$fontsize$$else$11pt$endif$,openany,twoside]{memoir}

  {$if(margin-outer)$$margin-outer$$else$14mm$endif$}{*}

\setmainfont{$if(font-main)$$font-main$$else$PT Serif$endif$}[Ligatures=TeX]

\frenchspacing

\clubpenalty=10000
\widowpenalty=10000
\raggedbottom

% ---------- Колонтитулы twoside ----------
\makepagestyle{book}
\makeevenhead{book}{\thepage}{}{\small\itshape \leftmark}
\makeoddhead{book}{\small\itshape \rightmark}{}{\thepage}
\makeevenfoot{book}{}{}{}
\makeoddfoot{book}{}{}{}

\chapterstyle{hangnum}

\renewcommand{\chapnumfont}{\normalfont\bfseries\Large}

\usepackage{tikz}
\usetikzlibrary{positioning, arrows.meta, shapes.geometric, fit, backgrounds}
\tikzset{
  opnode/.style={draw=black, line width=0.4pt, rectangle, rounded corners=2pt,
    inner sep=4pt, font=\footnotesize, align=center, minimum height=18pt},
  srcnode/.style={draw=black, line width=0.4pt, rectangle, fill=black!8,

    inner sep=4pt, font=\footnotesize\bfseries, align=center, minimum height=18pt},

}

% ---------- Подписи к рисункам и листингам ----------
